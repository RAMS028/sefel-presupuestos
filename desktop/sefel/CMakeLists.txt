cmake_minimum_required(VERSION 3.21)
project(SefelDesktop VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Sql)
qt_add_executable(sefel-desktop src/main.cpp src/mainwindow.cpp src/mainwindow.h src/database.cpp src/database.h)
target_link_libraries(sefel-desktop PRIVATE Qt6::Widgets Qt6::Sql)
